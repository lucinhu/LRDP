cmake_minimum_required(VERSION 3.22)
project(lrdp_native LANGUAGES C CXX)
set(OPENSSL_USE_STATIC_LIBS TRUE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../src" "${CMAKE_CURRENT_BINARY_DIR}/shared")
set(PLUGIN_NAME "lrdp_native_plugin")
add_library(${PLUGIN_NAME} SHARED lrdp_native_plugin.cpp video_texture.cc)
apply_standard_settings(${PLUGIN_NAME})
target_compile_definitions(${PLUGIN_NAME} PRIVATE LRDP_NATIVE_PLUGIN_IMPL)
target_include_directories(${PLUGIN_NAME} INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(${PLUGIN_NAME} PRIVATE flutter flutter_wrapper_plugin lrdp_core)
set(lrdp_native_bundled_libraries $<TARGET_FILE:lrdp_core> PARENT_SCOPE)
