cmake_minimum_required(VERSION 3.22)
project(lrdp_native VERSION 0.1.0 LANGUAGES C CXX)
if(CMAKE_GENERATOR MATCHES "Ninja" AND DEFINED ENV{LRDP_BUILD_JOBS})
  if(NOT "$ENV{LRDP_BUILD_JOBS}" MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "LRDP_BUILD_JOBS must be a positive integer")
  endif()
  set_property(GLOBAL APPEND PROPERTY JOB_POOLS "lrdp_compile=$ENV{LRDP_BUILD_JOBS}")
  set(CMAKE_JOB_POOL_COMPILE lrdp_compile)

option(LRDP_USE_SYSTEM_FREERDP "Use installed FreeRDP 3 (development only)" OFF)
option(LRDP_BUILD_TESTS "Build LRDP bridge tests" OFF)
option(LRDP_WITH_H264 "Decode RDP AVC420/AVC444 using FFmpeg" ON)
option(LRDP_WITH_AUDIO "Build the platform audio playback backend" ON)
set(LRDP_FFMPEG_ROOT "$ENV{LRDP_FFMPEG_ROOT}" CACHE PATH "Static FFmpeg prefix (Android: parent of ABI prefixes)")
if(DEFINED ENV{LRDP_FFMPEG_ROOT} AND NOT "$ENV{LRDP_FFMPEG_ROOT}" STREQUAL "")
  set(LRDP_FFMPEG_ROOT "$ENV{LRDP_FFMPEG_ROOT}" CACHE PATH "Static FFmpeg prefix" FORCE)
endif()
if(LRDP_WITH_H264 AND LRDP_FFMPEG_ROOT)
  set(lrdp_ffmpeg_prefix "${LRDP_FFMPEG_ROOT}")
  if(ANDROID)
    string(APPEND lrdp_ffmpeg_prefix "/${ANDROID_ABI}")
  endif()
  foreach(required include/libavcodec/avcodec.h include/libswscale/swscale.h)
    if(NOT EXISTS "${lrdp_ffmpeg_prefix}/${required}")
      message(FATAL_ERROR "Missing FFmpeg: ${lrdp_ffmpeg_prefix}/${required}; see scripts/build_ffmpeg.sh")
    endif()
  endforeach()
  set(FFMPEG_INCLUDE_DIRS "${lrdp_ffmpeg_prefix}/include" CACHE STRING "" FORCE)
  set(lrdp_ffmpeg_archives "")
  foreach(component AVCODEC SWSCALE AVUTIL)
    string(TOLOWER "${component}" library_name)
    set(archive_path "")
    foreach(filename "lib${library_name}.a" "${library_name}.lib" "lib${library_name}.lib")
      if(EXISTS "${lrdp_ffmpeg_prefix}/lib/${filename}")
        set(archive_path "${lrdp_ffmpeg_prefix}/lib/${filename}")
        break()
      endif()
    endforeach()
    if(NOT archive_path)
      message(FATAL_ERROR "Missing static ${library_name} archive under ${lrdp_ffmpeg_prefix}/lib")
    endif()
    list(APPEND lrdp_ffmpeg_archives "${archive_path}")
    set(${component}_LIBRARIES "${archive_path}" CACHE STRING "" FORCE)
    set(${component}_INCLUDE_DIRS "${lrdp_ffmpeg_prefix}/include" CACHE STRING "" FORCE)
    set(${component}_FOUND TRUE)
    set(FFmpeg_${component}_FOUND TRUE)
    file(READ "${lrdp_ffmpeg_prefix}/include/lib${library_name}/version.h" version_text)
    if(EXISTS "${lrdp_ffmpeg_prefix}/include/lib${library_name}/version_major.h")
      file(READ "${lrdp_ffmpeg_prefix}/include/lib${library_name}/version_major.h" major_text)
      string(APPEND version_text "\n${major_text}")
    endif()
    foreach(part MAJOR MINOR MICRO)
      string(REGEX MATCH "LIB${component}_VERSION_${part}[ \t]+([0-9]+)" match "${version_text}")
      set(version_${part} "${CMAKE_MATCH_1}")
    endforeach()
    set(${component}_VERSION "${version_MAJOR}.${version_MINOR}.${version_MICRO}" CACHE STRING "" FORCE)
  endforeach()
  set(FFMPEG_LIBRARIES "${lrdp_ffmpeg_archives}" CACHE STRING "" FORCE)
  set(FFMPEG_DEFINITIONS "" CACHE STRING "" FORCE)
endif()

  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "No LTO in development builds" FORCE)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MINGW)
  # WinPR's GNU unwind helper needs POSIX dlfcn.h; use its Windows backend.
  set(USE_UNWIND OFF CACHE BOOL "Use native Windows stack traces" FORCE)
  include(CheckIncludeFile)
  check_include_file(threads.h LRDP_HAVE_C11_THREADS)
  if(NOT LRDP_HAVE_C11_THREADS)
    # Some MinGW CRTs omit the optional C11 threads library without defining
    # its feature macro. Let upstream select its existing Win32 TLS fallback.
    add_compile_definitions(__STDC_NO_THREADS__=1)
  endif()
endif()
if(ANDROID)
  if(NOT DEFINED ENV{LRDP_OPENSSL_ROOT})
    message(FATAL_ERROR "Set LRDP_OPENSSL_ROOT to per-ABI static OpenSSL builds. See docs/BUILDING.md")
  endif()
  set(OPENSSL_ROOT_DIR "$ENV{LRDP_OPENSSL_ROOT}/${ANDROID_ABI}")
  foreach(required include/openssl/ssl.h lib/libssl.a lib/libcrypto.a)
    if(NOT EXISTS "${OPENSSL_ROOT_DIR}/${required}")
      message(FATAL_ERROR "Missing Android ${ANDROID_ABI} OpenSSL: ${OPENSSL_ROOT_DIR}/${required}. Build that ABI or select --target-platform android-arm64. See docs/BUILDING.md")
    endif()
  endforeach()
  set(OPENSSL_INCLUDE_DIR "${OPENSSL_ROOT_DIR}/include" CACHE PATH "" FORCE)
  set(OPENSSL_CRYPTO_LIBRARY "${OPENSSL_ROOT_DIR}/lib/libcrypto.a" CACHE FILEPATH "" FORCE)
  set(OPENSSL_SSL_LIBRARY "${OPENSSL_ROOT_DIR}/lib/libssl.a" CACHE FILEPATH "" FORCE)
  set(OPENSSL_USE_STATIC_LIBS TRUE)
endif()

  set(LRDP_FREERDP_LIBRARIES PkgConfig::FREERDP)

  endif()

  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(WITH_CLIENT OFF CACHE BOOL "" FORCE)
  set(WITH_CLIENT_COMMON ON CACHE BOOL "" FORCE)
  set(WITH_CHANNELS ON CACHE BOOL "" FORCE)
  set(WITH_CLIENT_CHANNELS ON CACHE BOOL "" FORCE)
  foreach(opt WITH_SERVER WITH_SERVER_CHANNELS WITH_SHADOW WITH_SAMPLE WITH_X11
      WITH_WAYLAND WITH_CLIENT_SDL WITH_SDL WITH_MANPAGES WITH_DOCBOOK
      WITH_SWSCALE WITH_PULSE WITH_ALSA WITH_CUPS WITH_PCSC WITH_SMARTCARD_EMULATE
      WITH_FUSE WITH_FUSE3 WITH_AAD WITH_WEBVIEW WITH_CAIRO WITH_OPENH264
      WITH_JPEG WITH_GSM WITH_FAAC WITH_FAAD2 WITH_LAME WITH_SOXR WITH_OPUS
      WITH_DSP_FFMPEG WITH_FFMPEG_HWACCEL WITH_VAAPI WITH_GSSAPI WITH_KRB5
      WITH_SYSTEMD WITH_WINPR_TOOLS WITH_WINPR_TOOLS_CLI
      WITH_OSS WITH_SNDIO WITH_MACAUDIO WITH_IOSAUDIO WITH_OPENSLES WITH_AAUDIO WITH_WINMM
      WITH_SSE2 WITH_NEON WITH_AVX2 WITH_AVX512 WITH_BINARY_VERSIONING
      WITH_CCACHE WITH_CLANG_FORMAT BUILD_TESTING BUILD_TESTING_INTERNAL)
    set(${opt} OFF CACHE BOOL "" FORCE)
  endforeach()
  if(LRDP_WITH_AUDIO)
    if(ANDROID)
      set(WITH_OPENSLES ON CACHE BOOL "Android API 24 compatible playback" FORCE)
    elseif(WIN32)
      set(WITH_WINMM ON CACHE BOOL "Windows playback" FORCE)
    elseif(APPLE)
      if(IOS OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
        set(WITH_IOSAUDIO ON CACHE BOOL "iOS playback" FORCE)
      else()
        set(WITH_MACAUDIO ON CACHE BOOL "macOS playback" FORCE)
      endif()
    else()
      set(WITH_PULSE ON CACHE BOOL "PulseAudio/PipeWire playback" FORCE)
    endif()
  endif()
  set(WITH_FFMPEG ${LRDP_WITH_H264} CACHE BOOL "" FORCE)
  set(WITH_VIDEO_FFMPEG ${LRDP_WITH_H264} CACHE BOOL "" FORCE)
  foreach(channel URBDRC TSMF AUDIN RAIL REMDESK ENCOMSP VIDEO
      GEOMETRY LOCATION ECHO AINPUT RDPEI RDPEMSC RDPECAM TELEMETRY)
    set(CHANNEL_${channel} OFF CACHE BOOL "" FORCE)
  endforeach()
  foreach(channel CLIPRDR DISP DRDYNVC RDPGFX RDPDR RDPSND DRIVE)
    set(CHANNEL_${channel} ON CACHE BOOL "" FORCE)
  endforeach()
  set(WITH_UNICODE_BUILTIN ON CACHE BOOL "" FORCE)
  set(WITH_JSON_DISABLED ON CACHE BOOL "" FORCE)
  add_subdirectory("${FREERDP_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/freerdp" EXCLUDE_FROM_ALL)
  if(MINGW AND CMAKE_C_COMPILER_ID STREQUAL "GNU")
    # Upstream 3.32 passes DWORD* to UINT32* in its legacy keyboard remapper.
    # Both are 32-bit on Win64. Retain the warning and MSVC-style aliasing for
    # this one source; keep the LRDP adapter's diagnostics strict.
    set_property(SOURCE "${FREERDP_ROOT}/libfreerdp/locale/keyboard.c"
      TARGET_DIRECTORY freerdp APPEND PROPERTY COMPILE_OPTIONS
      -Wno-error=incompatible-pointer-types -fno-strict-aliasing)
  endif()
  # FreeRDP enables this cache policy globally; Flutter generates platform
  # headers in its ephemeral source directory. Restore Flutter's policy.
  set(CMAKE_DISABLE_SOURCE_CHANGES OFF CACHE INTERNAL "Flutter generated headers" FORCE)
  set(CMAKE_DISABLE_IN_SOURCE_BUILD OFF CACHE INTERNAL "Flutter ephemeral build" FORCE)
  set(LRDP_FREERDP_LIBRARIES freerdp-client freerdp winpr)
endif()
add_library(lrdp_core SHARED src/lrdp.c src/audio_gate.c src/audio_buffer.c src/video_store.c)
if(UNIX AND NOT APPLE)
  # Static codec assembly uses local PC-relative tables. Keep archive symbols
  # private to this bridge; the explicitly exported LRDP C ABI remains public.
  target_link_options(lrdp_core PRIVATE "LINKER:--exclude-libs,ALL" "LINKER:-Bsymbolic-functions")
endif()
target_compile_definitions(lrdp_core PRIVATE LRDP_BUILDING_LIBRARY)
target_compile_definitions(lrdp_core PRIVATE LRDP_ENABLE_H264=$<BOOL:${LRDP_WITH_H264}>)
target_compile_definitions(lrdp_core PRIVATE LRDP_ENABLE_AUDIO=$<BOOL:${LRDP_WITH_AUDIO}>)
target_include_directories(lrdp_core PUBLIC include)
if(NOT LRDP_USE_SYSTEM_FREERDP)
  target_include_directories(lrdp_core SYSTEM PRIVATE
    "${FREERDP_ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}/freerdp/include"
    "${FREERDP_ROOT}/winpr/include" "${CMAKE_CURRENT_BINARY_DIR}/freerdp/winpr/include")
endif()
target_link_libraries(lrdp_core PRIVATE ${LRDP_FREERDP_LIBRARIES})
if(LRDP_WITH_H264 AND LRDP_FFMPEG_ROOT)
  if(WIN32)
    target_link_libraries(lrdp_core PRIVATE bcrypt)
  else()
    target_link_libraries(lrdp_core PRIVATE m)
  endif()
endif()
target_compile_features(lrdp_core PRIVATE c_std_11)
set_target_properties(lrdp_core PROPERTIES OUTPUT_NAME lrdp C_VISIBILITY_PRESET hidden
  BUILD_RPATH_USE_ORIGIN ON INSTALL_RPATH "$ORIGIN")
if(WIN32)
  # The Dart loader always opens lrdp.dll, including with MinGW toolchains.
  set_target_properties(lrdp_core PROPERTIES PREFIX "")
  target_link_libraries(lrdp_core PRIVATE ws2_32 shlwapi)
  if(MINGW)
    # FFmpeg's compiler support can require Winpthreads even with its Win32
    # video threads. Embed toolchain runtimes alongside the static codecs.
    target_link_options(lrdp_core PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  if(NOT LRDP_USE_SYSTEM_FREERDP)
    # Upstream's Windows headers use export/import decoration even for static
    # archives. These APIs are being embedded into this DLL, not imported.
    target_compile_definitions(lrdp_core PRIVATE FREERDP_EXPORTS WINPR_EXPORTS)
  endif()
endif()
